    <script src="https://unpkg.com/react@18/umd/react.production.min.js"></script>

    <script src="https://unpkg.com/react-dom@18/umd/react-dom.production.min.js"></script>

    <script src="https://unpkg.com/@babel/standalone/babel.min.js"></script>

    <script src="https://cdn.tailwindcss.com"></script>

    <script src="https://unpkg.com/lucide@latest"></script>

    <style>

        @keyframes brushSpinVertical { 0% { transform: scaleX(1); } 25% { transform: scaleX(0.05); } 50% { transform: scaleX(1); } 75% { transform: scaleX(0.05); } 100% { transform: scaleX(1); } }

        @keyframes fullWaterfall { 0% { transform: translateY(-100%); } 100% { transform: translateY(100%); } }

        @keyframes bubbleFloat { 0%, 100% { transform: translateY(0) translateX(0); } 50% { transform: translateY(-20px) translateX(10px); } }

        @keyframes twinkle { 0%, 100% { opacity: 0; transform: scale(0.5) rotate(0deg); } 50% { opacity: 1; transform: scale(1.2) rotate(180deg); } }

        @keyframes jitter { 0% { transform: translateX(-50%) translate(0, 0); } 10% { transform: translateX(-50%) translate(-2px, -2px); } 20% { transform: translateX(-50%) translate(2px, 2px); } 100% { transform: translateX(-50%) translate(0, 0); } }

        @keyframes bounce { 0%, 100% { transform: translateX(-50%) translateY(0) scale(1.1); } 50% { transform: translateX(-50%) translateY(-10px) scale(1.1); } }

        .animate-twinkle { animation: twinkle 1.5s infinite ease-in-out; }

<body>

    <div id="root"></div>


    <script type="text/babel">

        const { useState, useRef, useEffect } = React;

        

        // Icons

        const Waves = () => <i data-lucide="waves"></i>;

        const RotateCcw = () => <i data-lucide="rotate-ccw"></i>;

        const Droplets = () => <i data-lucide="droplets"></i>;

        const Info = () => <i data-lucide="info"></i>;

        const GripHorizontal = () => <i data-lucide="grip-horizontal"></i>;

        const Smile = ({ className }) => <i data-lucide="smile" className={className}></i>;

        const Frown = ({ className }) => <i data-lucide="frown" className={className}></i>;

        const Star = ({ size }) => <i data-lucide="star" style={{width: size, height: size}}></i>;


        const App = () => {

            const [progress, setProgress] = useState(0);

            const [isDragging, setIsDragging] = useState(false);

            const [showSparkles, setShowSparkles] = useState(false);

            const trackRef = useRef(null);


            // Replace with your final IDs if different

            const rustedImg = "https://lh3.googleusercontent.com/u/0/d/1yPL_3ymGeOvoXzECwC8psNnwvT8e-03G";

            const restoredImg = "https://lh3.googleusercontent.com/u/0/d/1mLtjbjGBxbJeoxFbdgltRPKxiLWmaMYC";


            useEffect(() => { lucide.createIcons(); }, []);


            const handleMove = (clientX) => {

                if (!isDragging || !trackRef.current) return;

                const rect = trackRef.current.getBoundingClientRect();

                const x = clientX - rect.left;

                const newProgress = Math.max(0, Math.min(100, (x / rect.width) * 100));

                setProgress(newProgress);

                setShowSparkles(newProgress > 85);

            };


            useEffect(() => {

                const onMouseMove = (e) => handleMove(e.clientX);

                const onEnd = () => setIsDragging(false);

                if (isDragging) {

                    window.addEventListener('mousemove', onMouseMove);

                    window.addEventListener('mouseup', onEnd);

                }

                return () => {

                    window.removeEventListener('mousemove', onMouseMove);

                    window.removeEventListener('mouseup', onEnd);

                };

            }, [isDragging]);


            const getTransitionProgress = () => {

                const start = 35, end = 65;

                if (progress < start) return 0;

                if (progress > end) return 100;

                return ((progress - start) / (end - start)) * 100;

            };


            const transitionVal = getTransitionProgress();

            const isInWash = progress >= 32 && progress <= 68;

            const blingPositions = [{t:'20%',l:'30%'},{t:'40%',l:'70%'},{t:'15%',l:'50%'},{t:'60%',l:'20%'},{t:'55%',l:'80%'}];


            return (

                <div className="min-h-screen bg-white flex flex-col items-center justify-center p-8 font-sans select-none">

                    <div className="max-w-5xl w-full mb-8 flex justify-between items-end px-4">

                        <div>

                            <h1 className="text-3xl font-black text-slate-900 flex items-center gap-3 italic tracking-tighter uppercase">

                                <Waves /> PowerAccred Car Wash

                            </h1>

                            <p className="text-slate-400 font-semibold text-[10px] tracking-[0.2em] uppercase">High-Pressure Restoration & Morale Boost</p>

                        </div>

                        <button onClick={() => setProgress(0)} className="flex items-center gap-2 px-5 py-2 bg-slate-50 hover:bg-blue-500 hover:text-white text-slate-600 rounded-full transition-all border border-slate-200 font-bold text-sm shadow-sm">

                            <RotateCcw /> RESET

                        </button>

                    </div>


                    <div ref={trackRef} className="max-w-6xl w-full aspect-[21/9] relative bg-slate-50 rounded-[4rem] border-[16px] border-white shadow-[0_20px_50px_rgba(0,0,0,0.05)] overflow-hidden">

                        <div className="absolute top-1/2 -translate-y-1/2 w-[30%] aspect-video z-10 transition-all duration-150" style={{ left: `calc(${progress}% - ${progress * 0.30}%)` }}>

                            <div className="relative w-full h-full pointer-events-none">

                                {/* Frustration Bubble */}

                                <div className={`absolute -top-24 left-1/2 -translate-x-1/2 z-30 transition-all duration-500 ${transitionVal < 5 ? 'animate-jitter' : ''}`} style={{ opacity: transitionVal < 5 ? 1 : 0 }}>

                                    <svg width="120" height="100" viewBox="0 0 120 100">

                                        <path d="M10,40 L30,10 L60,0 L90,10 L110,40 L90,70 L60,85 L30,70 Z" fill="white" stroke="#ef4444" strokeWidth="3" />

                                        <path d="M35,45 C30,30 45,20 60,35 C75,50 45,70 30,55 C15,40 50,25 75,40 C100,55 60,75 40,65 C20,55 50,40 70,55" fill="none" stroke="black" strokeWidth="2.5" />

                                    </svg>

                                </div>

                                {/* Excitement Bubble */}

                                <div className={`absolute -top-28 left-1/2 -translate-x-1/2 z-30 transition-all duration-500 ${transitionVal > 95 ? 'animate-bounce' : ''}`} style={{ opacity: transitionVal > 95 ? 1 : 0 }}>

                                    <svg width="140" height="110" viewBox="0 0 140 110">

                                        <path d="M10,50 L30,40 L25,20 L50,30 L60,5 L75,30 L100,20 L95,45 L120,55 L95,65 L105,90 L75,80 L60,105 L45,80 L15,90 L25,65 Z" fill="white" stroke="#10b981" strokeWidth="3" />

                                        <text x="50%" y="55%" textAnchor="middle" fill="#065f46" style={{ fontSize: '18px', fontWeight: 900 }}>OH YEAH!!</text>

                                    </svg>

                                </div>

                                <img src={restoredImg} className="absolute inset-0 w-full h-full object-contain" style={{ clipPath: `inset(0 0 0 ${100 - transitionVal}%)` }} />

                                <img src={rustedImg} className="absolute inset-0 w-full h-full object-contain" style={{ clipPath: `inset(0 ${transitionVal}% 0 0)` }} />

                                {showSparkles && blingPositions.map((pos, i) => (

                                    <div key={i} className="absolute animate-twinkle text-yellow-400" style={{ top: pos.t, left: pos.l, animationDelay: `${i*0.4}s` }}><Star size={20} /></div>

                                ))}

                            </div>

                            <div className={`absolute -bottom-14 left-1/2 -translate-x-1/2 p-4 rounded-full border-2 cursor-grab active:cursor-grabbing transition-all ${isDragging ? 'bg-blue-600 border-blue-400' : 'bg-white border-slate-200'}`} onMouseDown={() => setIsDragging(true)}>

                                <GripHorizontal />

                            </div>

                        </div>


                        <div className="absolute top-0 bottom-0 left-[35%] right-[35%] z-20 pointer-events-none">

                            <div className="absolute inset-0 bg-blue-500/5 backdrop-blur-[15px] border-x-[20px] border-white shadow-[inset_0_0_100px_rgba(59,130,246,0.3)]">

                                <div className="absolute inset-0 overflow-hidden mix-blend-multiply opacity-40">

                                    {[...Array(30)].map((_, i) => <div key={i} className="absolute top-0 w-[4px] bg-blue-400 rounded-full blur-[2px]" style={{ left: `${i*3.3}%`, height:'100%', animation: `fullWaterfall 0.4s infinite linear`, animationDelay: `${i*0.05}s` }} />)}

                                </div>

                                <div className={`absolute inset-0 transition-opacity duration-500 ${isInWash ? 'opacity-100' : 'opacity-10'}`}>

                                    {[...Array(50)].map((_, i) => <div key={i} className="absolute bg-white/80 rounded-full blur-[20px] animate-pulse" style={{ width: '80px', height:'80px', top: `${Math.random()*100}%`, left: `${Math.random()*100}%` }} />)}

                                </div>

                                <div className="absolute inset-0 flex justify-around items-center px-4 py-24 opacity-50">

                                    {[...Array(3)].map((_, i) => <div key={i} className="h-full w-20 relative" style={{ animation: 'brushSpinVertical 0.5s infinite linear' }}>

                                        {[...Array(10)].map((_, j) => <div key={j} className="absolute w-full h-4 bg-yellow-400 rounded-full blur-[1px]" style={{ top: `${j*10}%`, transform: `rotate(${j*36}deg) scaleX(1.2)` }} />)}

                                    </div>)}

                                </div>

                            </div>

                        </div>

                    </div>


                    <div className="max-w-2xl w-full mt-10 bg-slate-50 border border-slate-100 p-8 rounded-[3rem] shadow-sm text-center">

                        <div className="flex justify-between items-center mb-6">

                            <div className="flex items-center gap-3 text-left">

                                <div className={`p-3 rounded-2xl ${isInWash ? 'bg-blue-600 text-white' : 'bg-white border border-slate-100'}`}>

                                    {transitionVal < 50 ? <Frown /> : <Smile className="text-white" />}

                                </div>

                                <div><span className="text-slate-900 text-[10px] font-black uppercase tracking-widest block">Condition Status</span></div>

                            </div>

                            <span className="text-4xl font-black font-mono text-slate-900">{Math.round(progress)}%</span>

                        </div>

                        <div className="w-full h-8 bg-white border border-slate-100 rounded-full p-2 shadow-inner">

                            <div className="h-full bg-gradient-to-r from-blue-600 to-emerald-400 rounded-full transition-all duration-300" style={{ width: `${progress}%` }} />

                        </div>

                    </div>

                </div>

            );

        };


        const root = ReactDOM.createRoot(document.getElementById('root'));

        root.render(<App />);

    </script>
